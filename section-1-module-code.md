  RealColumn get pctToday => real()();

/// Failure-shield ledger. The order requires every state-changing op
/// to carry a key and a repeated key to be a silent no-op. The primary
/// key is the COMPOSITE {op, idempotencyKey} so that the SAME key
/// string used by two different operations is treated as two distinct
/// writes, not one replay (Fix C, v1.0.1).
class IdempotencyLedger extends Table {
  TextColumn get idempotencyKey => text()();
  TextColumn get op => text()();
  DateTimeColumn get createdAt => dateTime()();

  @override
  Set<Column> get primaryKey => {op, idempotencyKey};
}

/// The database handle. `schemaVersion` is pinned to the contract
/// version; PRAGMAs are applied on every open so WAL, the busy
/// timeout and foreign-key checks are always on.
@DriftDatabase(
  tables: [Categories, Books, ProgressLog, DailySummary, IdempotencyLedger],
)
class M1Database extends _$M1Database {
  M1Database(super.e);

  /// Tests use a throwaway in-memory DB; production uses the file DB.
  M1Database.memory() : super(NativeDatabase.memory());

  @override
  int get schemaVersion => kSchemaVersion;

  @override
  MigrationStrategy get migration => MigrationStrategy(
        // v1.0.1 assumes a FRESH database: M1 has not shipped, so the
        // ledger's composite key needs no upgrade path yet. A released
        // upgrade would add an onUpgrade step here.
        onCreate: (Migrator m) async => m.createAll(),
        beforeOpen: (OpeningDetails details) async {
          await customStatement('PRAGMA journal_mode=WAL');
          await customStatement('PRAGMA busy_timeout=$kBusyTimeoutMs');
          await customStatement('PRAGMA foreign_keys=ON');
        },
      );
}

// === ZONE 2: BOUNDARY VALIDATION & MACHINE SOCKET CONTRACT ===
// Everything a caller can hand in is checked HERE, at the door,
// before any business rule or database work runs. Every failure
// leaves through one of the declared error shapes E1..E7.
// ============================================================

/// The one entry point. Takes an operation envelope, returns a
/// success or failure map. Never throws for a contract violation —
/// only a genuine storage fault becomes E7.
class M1LibraryStore {
  M1LibraryStore(this.db);

  final M1Database db;

  /// Ops that change stored data and therefore demand a key.
  static const Set<String> _stateChangingOps = <String>{
    'add_book',
    'update_progress',
  };

  static const Set<String> _knownOps = <String>{
    'upsert_category',
    'add_book',
    'update_progress',
    'read_catalog',
    'read_daily',
    'delete_book',
  };

  /// Envelope keys a caller may send; anything else is E1.
  static const Set<String> _envelopeKeys = <String>{
    'op',
    'payload',
    'idempotency_key',
    'schema_version',
  };

  /// Per-op payload key sets: which keys are allowed, which must be
  /// present. A missing or extra key is E1 ("missing/extra field").
  static const Map<String, Set<String>> _allowedPayloadKeys = <String, Set<String>>{
    'upsert_category': {'category_id', 'name', 'position'},
    'add_book': {
      'book_id',
      'category_id',
      'title',
      'total_pages',
      'source',
      'file_path',
      'cover_ref',
    },
    'update_progress': {'book_id', 'current_page', 'read_on'},
    'read_catalog': <String>{},
    'read_daily': {'as_of_date'},
    'delete_book': {'book_id'},
  };

  static const Map<String, Set<String>> _requiredPayloadKeys = <String, Set<String>>{
    'upsert_category': {'category_id', 'name', 'position'},
    'add_book': {
      'book_id',
      'category_id',
      'title',
      'total_pages',
      'source',
      'file_path',
      'cover_ref',
    },
    'update_progress': {'book_id', 'current_page', 'read_on'},
    'read_catalog': <String>{},
    'read_daily': {'as_of_date'},
    'delete_book': {'book_id'},
  };

  /// THE socket. Validates the envelope, enforces idempotency, then
  /// dispatches to the op handler. All storage faults funnel to E7.
  Future<Map<String, dynamic>> handle(Map<String, dynamic> envelope) async {
    // --- envelope-level boundary checks -----------------------------
    for (final String k in envelope.keys) {
      if (!_envelopeKeys.contains(k)) {
        return _fail('E1', 'Unexpected envelope field "$k".', k);
      }
    }
    if (!envelope.containsKey('op') ||
        !envelope.containsKey('payload') ||
        !envelope.containsKey('schema_version')) {
      return _fail('E1', 'Envelope is missing op, payload or schema_version.');
    }

    final Object? op = envelope['op'];
    if (op is! String || !_knownOps.contains(op)) {
      return _fail('E1', 'Unknown or missing op.', 'op');
    }

    final Object? sv = envelope['schema_version'];
    if (sv is! int || sv != kSchemaVersion) {
      return _fail('E1', 'schema_version must equal $kSchemaVersion.',
          'schema_version');
    }

    final Object? payloadRaw = envelope['payload'];
    if (payloadRaw is! Map) {
      return _fail('E1', 'payload must be an object.', 'payload');
    }
    final Map<String, dynamic> payload =
        Map<String, dynamic>.from(payloadRaw as Map);

    // Payload key discipline: unknown key or missing required key -> E1.
    final Set<String> allowed = _allowedPayloadKeys[op]!;
    final Set<String> required = _requiredPayloadKeys[op]!;
    for (final String k in payload.keys) {
      if (!allowed.contains(k)) {
        return _fail('E1', 'Unexpected payload field "$k" for $op.', k);
      }
    }
    for (final String k in required) {
      if (!payload.containsKey(k)) {
        return _fail('E1', 'Missing payload field "$k" for $op.', k);
      }
    }

    // State-changing ops must carry a usable key.
    final Object? idemRaw = envelope['idempotency_key'];
    String? idemKey;
    if (_stateChangingOps.contains(op)) {
      if (idemRaw is! String || idemRaw.isEmpty) {
        return _fail(
            'E1', 'idempotency_key is required for $op.', 'idempotency_key');
      }
      idemKey = idemRaw;
    } else if (idemRaw is String && idemRaw.isNotEmpty) {
      idemKey = idemRaw;
    }

    // --- dispatch, with the storage shield wrapped around it --------
    try {
      if (idemKey != null && _stateChangingOps.contains(op)) {
        final Map<String, dynamic>? replay = await _replayIfSeen(op, payload, idemKey);
        if (replay != null) return replay;
      }
      switch (op) {
        case 'upsert_category':
          return await _upsertCategory(payload);
        case 'add_book':
          return await _addBook(payload, idemKey);
        case 'update_progress':
          return await _updateProgress(payload, idemKey!);
        case 'read_catalog':
          return await _readCatalog();
        case 'read_daily':
          return await _readDaily(payload);
        case 'delete_book':
          return await _deleteBook(payload);
        default:
          // Unreachable: op was checked against _knownOps above.
          return _fail('E1', 'Unknown op.', 'op');
      }
    } on SqliteException catch (e) {
      return _storageError(e.message);
    } catch (e) {
      return _storageError(e.toString());
    }
  }

  // ============================================================
  // === ZONE 3: CONTROL PATH FUNCTIONS (THE SWITCHES) ===
  // Pure decisions and business rules. No SQL, no formatting, no
  // transport — so a rule can be read and changed in isolation.
  // ============================================================

  /// C1 (launch). On first launch there are no categories, so seed
  /// exactly kMaxCategories empty tiles named "Category 1".."N" so
  /// the home grid is never blank. Idempotent: a second call is a
  /// no-op because categories already exist.
  Future<void> seedIfEmpty() async {
    final int count = await _countCategories();
    if (count != 0) return;
    await db.transaction(() async {
      for (int i = 0; i < kMaxCategories; i++) {
        await db.into(db.categories).insert(
              CategoriesCompanion.insert(
                categoryId: _uuidV4(),
                name: 'Category ${i + 1}',
                position: i,
              ),
            );
      }
    });
  }

  /// C2 (add book). Decide whether one more book fits. Returns true
  /// only while the category holds fewer than the cap.
  bool categoryHasRoom(int currentCount) => currentCount < kMaxBooksPerCategory;

  /// C2 companion. Decide whether one more category may be created.
  bool catalogHasRoom(int currentCount) => currentCount < kMaxCategories;

  /// True only for a well-formed calendar date in "YYYY-MM-DD".
  bool isIsoDate(String value) {
    final RegExp re = RegExp(r'^\d{4}-\d{2}-\d{2}$');
    if (!re.hasMatch(value)) return false;
    try {
      final DateTime parsed = DateTime.parse(value);
      // DateTime.parse is lenient; re-format and compare to reject
      // impossible dates such as 2026-02-31.
      final String round =
          '${parsed.year.toString().padLeft(4, '0')}-'
          '${parsed.month.toString().padLeft(2, '0')}-'
          '${parsed.day.toString().padLeft(2, '0')}';
      return round == value;
    } catch (_) {
      return false;
    }
  }

  /// Pages read on one date for one book = (highest page recorded on
  /// that date) minus (highest page recorded on any earlier date,
  /// or 0 if this is the first reading). Never negative.
  int pagesReadOn({required int maxOnDay, required int maxBefore}) {
    final int delta = maxOnDay - maxBefore;
    return delta < 0 ? 0 : delta;
  }

  // ============================================================
  // === ZONE 4: DATA PATH, FAILURE-SHIELD & HEALTH-PROBE (PIPES) ===
  // Idempotency ledger, transactional writes, the daily roll-up,
  // and the integrity probe. Formatting/transport only — all
  // decisions already happened in Zone 3.
  // ============================================================

  /// Replay guard. If this key was already used by this op, do NOT
  /// write again: return the exact success shape the caller expects.
  /// Returns null when the key is new (caller should proceed).
  Future<Map<String, dynamic>?> _replayIfSeen(
    String op,
    Map<String, dynamic> payload,
    String idemKey,
  ) async {
    // Namespaced by op: a replay is only a replay for the SAME operation.
    final IdempotencyLedgerData? seen = await (db.select(db.idempotencyLedger)
          ..where((t) => t.op.equals(op) & t.idempotencyKey.equals(idemKey)))
        .getSingleOrNull();
    if (seen == null) return null;

    if (op == 'update_progress') {
      final String bookId = payload['book_id'] as String;
      final Book? book = await (db.select(db.books)
            ..where((t) => t.bookId.equals(bookId)))
          .getSingleOrNull();
      final ProgressLogData? last = await (db.select(db.progressLog)
            ..where((t) => t.bookId.equals(bookId))
            ..orderBy([(t) => OrderingTerm.desc(t.createdAt)])
            ..limit(1))
          .getSingleOrNull();
      final int page = last?.currentPage ?? (payload['current_page'] as int);
      final int total = book?.totalPages ?? 1;
      final double pct = total == 0 ? 0.0 : (page / total) * 100.0;
      return _ok({
        'book_id': bookId,
        'current_page': page,
        'pct_read': pct,
        'saved': false,
      });
    }

    // add_book replay: report the book as it already stands.
    final String bookId = payload['book_id'] as String;
    final Book? book = await (db.select(db.books)
          ..where((t) => t.bookId.equals(bookId)))
        .getSingleOrNull();
    if (book == null) {
      return _fail('E4', 'book_id does not exist.', 'book_id');
    }
    final int count = await _countBooksInCategory(book.categoryId);
    return _ok({
      'book_id': book.bookId,
      'category_id': book.categoryId,
      'book_count': count,
    });
  }

  /// upsert_category: create or update a tile. A NEW tile when the
  /// catalog is full is E3; an existing tile may always be renamed.
  Future<Map<String, dynamic>> _upsertCategory(
      Map<String, dynamic> payload) async {
    final Object? categoryId = payload['category_id'];
    if (categoryId is! String || categoryId.isEmpty) {
      return _fail('E1', 'category_id must be a non-empty string.', 'category_id');
    }
    final Object? name = payload['name'];
    if (name is! String || name.length < kNameMin || name.length > kNameMax) {
      return _fail('E1', 'name must be $kNameMin-$kNameMax characters.', 'name');
    }
    final Object? position = payload['position'];
    if (position is! int) {
      return _fail('E1', 'position must be an integer.', 'position');
    }
    if (position < kPositionMin || position > kPositionMax) {
      return _fail('E2', 'position must be $kPositionMin-$kPositionMax.', 'position');
    }

    final Category? existing = await (db.select(db.categories)
          ..where((t) => t.categoryId.equals(categoryId)))
        .getSingleOrNull();
    if (existing == null) {
      final int count = await _countCategories();
      if (!catalogHasRoom(count)) {
        return _fail('E3', 'Maximum of $kMaxCategories categories reached.');
      }
    }

    await db.into(db.categories).insertOnConflictUpdate(
          CategoriesCompanion.insert(
            categoryId: categoryId,
            name: name,
            position: position,
          ),
        );
    return _ok({'category_id': categoryId});
  }

  /// add_book: validate, then insert inside one transaction. Category
  /// must exist (E4) and have room (E3); the book_id must be new (E5).
  Future<Map<String, dynamic>> _addBook(
    Map<String, dynamic> payload,
    String? idemKey,
  ) async {
    final Object? bookId = payload['book_id'];
    if (bookId is! String || bookId.isEmpty) {
      return _fail('E1', 'book_id must be a non-empty string.', 'book_id');
    }
    final Object? categoryId = payload['category_id'];
    if (categoryId is! String || categoryId.isEmpty) {
      return _fail('E1', 'category_id must be a non-empty string.', 'category_id');
    }
    final Object? title = payload['title'];
    if (title is! String || title.length < kTitleMin || title.length > kTitleMax) {
      return _fail('E1', 'title must be $kTitleMin-$kTitleMax characters.', 'title');
    }
    final Object? totalPages = payload['total_pages'];
    if (totalPages is! int) {
      return _fail('E1', 'total_pages must be an integer.', 'total_pages');
    }
    if (totalPages < kTotalPagesMin || totalPages > kTotalPagesMax) {
      return _fail('E2', 'total_pages must be $kTotalPagesMin-$kTotalPagesMax.',
          'total_pages');
    }
    final Object? source = payload['source'];
    if (source is! String || !kSources.contains(source)) {
      return _fail('E1', 'source must be one of $kSources.', 'source');
    }
    final Object? filePathRaw = payload['file_path'];
    if (filePathRaw != null && filePathRaw is! String) {
      return _fail('E1', 'file_path must be a string or null.', 'file_path');
    }
    final String? filePath = filePathRaw as String?;
    if (source == 'device_file' && (filePath == null || filePath.isEmpty)) {
      return _fail('E1', 'file_path is required when source is device_file.',
          'file_path');
    }
    final Object? coverRefRaw = payload['cover_ref'];
    if (coverRefRaw != null && coverRefRaw is! String) {
      return _fail('E1', 'cover_ref must be a string or null.', 'cover_ref');
    }
    final String? coverRef = coverRefRaw as String?;

    final Category? category = await (db.select(db.categories)
          ..where((t) => t.categoryId.equals(categoryId)))
        .getSingleOrNull();
    if (category == null) {
      return _fail('E4', 'category_id does not exist.', 'category_id');
    }
    final int inCategory = await _countBooksInCategory(categoryId);
    if (!categoryHasRoom(inCategory)) {
      return _fail('E3', 'Category already holds $kMaxBooksPerCategory books.');
    }
    final Book? duplicate = await (db.select(db.books)
          ..where((t) => t.bookId.equals(bookId)))
        .getSingleOrNull();
    if (duplicate != null) {
      return _fail('E5', 'book_id already exists.', 'book_id');
    }

    await db.transaction(() async {
      await db.into(db.books).insert(
            BooksCompanion.insert(
              bookId: bookId,
              categoryId: categoryId,
              title: title,
              totalPages: totalPages,
              source: source,
              filePath: Value<String?>(filePath),
              coverRef: Value<String?>(coverRef),
            ),
          );
      if (idemKey != null) {
        await _rememberKey(idemKey, 'add_book');
      }
    });

    final int newCount = await _countBooksInCategory(categoryId);
    return _ok({
      'book_id': bookId,
      'category_id': categoryId,
      'book_count': newCount,
    });
  }

  /// update_progress: record a reading event and refresh the day's
  /// roll-up, atomically. Book must exist (E4); page must be within
  /// the book (E2); the date must be a real date (E1).
  Future<Map<String, dynamic>> _updateProgress(
    Map<String, dynamic> payload,
    String idemKey,
  ) async {
    final Object? bookId = payload['book_id'];
    if (bookId is! String || bookId.isEmpty) {
      return _fail('E1', 'book_id must be a non-empty string.', 'book_id');
    }
    final Object? currentPage = payload['current_page'];
    if (currentPage is! int) {
      return _fail('E1', 'current_page must be an integer.', 'current_page');
    }
    final Object? readOn = payload['read_on'];
    if (readOn is! String || !isIsoDate(readOn)) {
      return _fail('E1', 'read_on must match YYYY-MM-DD.', 'read_on');
    }

    final Book? book = await (db.select(db.books)
          ..where((t) => t.bookId.equals(bookId)))
        .getSingleOrNull();
    if (book == null) {
      return _fail('E4', 'book_id does not exist.', 'book_id');
    }
    if (currentPage < 0 || currentPage > book.totalPages) {
      return _fail('E2', 'current_page must be 0-${book.totalPages}.', 'current_page');
    }

    await db.transaction(() async {
      await db.into(db.progressLog).insert(
            ProgressLogCompanion.insert(
              bookId: bookId,
              currentPage: currentPage,
              readOn: readOn,
              idempotencyKey: idemKey,
              createdAt: DateTime.now(),
            ),
          );
      await _refreshDailySummary(bookId, readOn, book.totalPages);
      await _rememberKey(idemKey, 'update_progress');
    });

    final double pct =
        book.totalPages == 0 ? 0.0 : (currentPage / book.totalPages) * 100.0;
    return _ok({
      'book_id': bookId,
      'current_page': currentPage,
      'pct_read': pct,
      'saved': true,
    });
  }

  /// read_catalog: the whole grid in one shot — every category with
  /// its books, ordered by tile position.
  Future<Map<String, dynamic>> _readCatalog() async {
    final List<Category> cats = await (db.select(db.categories)
          ..orderBy([(t) => OrderingTerm.asc(t.position)]))
        .get();
    final List<Map<String, dynamic>> out = <Map<String, dynamic>>[];
    for (final Category c in cats) {
      final List<Book> books = await (db.select(db.books)
            ..where((t) => t.categoryId.equals(c.categoryId)))
          .get();
      out.add(<String, dynamic>{
        'category_id': c.categoryId,
        'name': c.name,
        'position': c.position,
        'books': books
            .map((Book b) => <String, dynamic>{
                  'book_id': b.bookId,
                  'title': b.title,
                  'total_pages': b.totalPages,
                  'source': b.source,
                  'cover_ref': b.coverRef,
                })
            .toList(),
      });
    }
    return _ok({'categories': out});
  }

  /// read_daily: the day's totals, read from the D4 roll-up so M4
  /// never has to re-aggregate the whole log.
  Future<Map<String, dynamic>> _readDaily(Map<String, dynamic> payload) async {
    final Object? asOf = payload['as_of_date'];
    if (asOf is! String || !isIsoDate(asOf)) {
      return _fail('E1', 'as_of_date must match YYYY-MM-DD.', 'as_of_date');
    }
    final List<DailySummaryData> rows = await (db.select(db.dailySummary)
          ..where((t) => t.date.equals(asOf)))
        .get();
    final List<Map<String, dynamic>> perBook = <Map<String, dynamic>>[];
    int total = 0;
    for (final DailySummaryData r in rows) {
      perBook.add(<String, dynamic>{
        'book_id': r.bookId,
        'pages_today': r.pagesToday,
        'pct_today': r.pctToday,
      });
      total += r.pagesToday;
    }
    return _ok({
      'date': asOf,
      'total_pages_today': total,
      'per_book': perBook,
    });
  }

  /// delete_book: remove a book and everything that hangs off it, in
  /// one transaction, so no orphan progress rows survive.
  Future<Map<String, dynamic>> _deleteBook(Map<String, dynamic> payload) async {
    final Object? bookId = payload['book_id'];
    if (bookId is! String || bookId.isEmpty) {
      return _fail('E1', 'book_id must be a non-empty string.', 'book_id');
    }
    final Book? book = await (db.select(db.books)
          ..where((t) => t.bookId.equals(bookId)))
        .getSingleOrNull();
    if (book == null) {
      return _fail('E4', 'book_id does not exist.', 'book_id');
    }
    await db.transaction(() async {
      await (db.delete(db.progressLog)..where((t) => t.bookId.equals(bookId))).go();
      await (db.delete(db.dailySummary)..where((t) => t.bookId.equals(bookId))).go();
      await (db.delete(db.books)..where((t) => t.bookId.equals(bookId))).go();
    });
    return _ok({'book_id': bookId, 'deleted': true});
  }

  /// Recompute one book's row in the D4 roll-up for one date.
  Future<void> _refreshDailySummary(
      String bookId, String date, int totalPages) async {
    final int maxOnDay = await _maxPageForBookOnDate(bookId, date);
    final int maxBefore = await _maxPageForBookBeforeDate(bookId, date);
    final int pages = pagesReadOn(maxOnDay: maxOnDay, maxBefore: maxBefore);
    final double pct = totalPages == 0 ? 0.0 : (pages / totalPages) * 100.0;
    await db.into(db.dailySummary).insertOnConflictUpdate(
          DailySummaryCompanion.insert(
            bookId: bookId,
            date: date,
            pagesToday: pages,
            pctToday: pct,
          ),
        );
  }

  Future<int> _maxPageForBookOnDate(String bookId, String date) async {
    final TypedResult row = await (db.selectOnly(db.progressLog)
          ..addColumns([db.progressLog.currentPage.max()])
          ..where(db.progressLog.bookId.equals(bookId) &
              db.progressLog.readOn.equals(date)))
        .getSingle();
    return row.read(db.progressLog.currentPage.max()) ?? 0;
  }

  Future<int> _maxPageForBookBeforeDate(String bookId, String date) async {
    final TypedResult row = await (db.selectOnly(db.progressLog)
          ..addColumns([db.progressLog.currentPage.max()])
          ..where(db.progressLog.bookId.equals(bookId) &
              db.progressLog.readOn.isSmallerThanValue(date)))
        .getSingle();
    return row.read(db.progressLog.currentPage.max()) ?? 0;
  }

  /// Health probe for M6. Returns the raw PRAGMA result: "ok" means
  /// the file is intact; anything else means "restore from backup".
  Future<String> healthCheck() async {
    final QueryRow row =
        await db.customSelect('PRAGMA integrity_check').getSingle();
    final Object? value = row.data.values.first;
    return value?.toString() ?? 'unknown';
  }

  /// Record a used idempotency key. UNIQUE means a genuine clash is a
  /// duplicate replay we already handled, so we swallow it quietly.
  Future<void> _rememberKey(String idemKey, String op) async {
    await db.into(db.idempotencyLedger).insert(
          IdempotencyLedgerCompanion.insert(
            idempotencyKey: idemKey,
            op: op,
            createdAt: DateTime.now(),
          ),
          mode: InsertMode.insertOrIgnore,
        );
  }

  Future<int> _countCategories() async {
    final QueryRow row = await db
        .customSelect('SELECT COUNT(*) AS c FROM categories')
        .getSingle();
    return row.data['c'] as int? ?? 0;
  }

  Future<int> _countBooksInCategory(String categoryId) async {
    final TypedResult row = await (db.selectOnly(db.books)
          ..addColumns([db.books.bookId.count()])
          ..where(db.books.categoryId.equals(categoryId)))

  }

  /// A dependency-free v4 UUID so seeded category ids are valid uuids
  /// without pulling in an extra package.
  static String _uuidV4() {
    final Random rnd = Random.secure();
    final List<int> bytes = List<int>.generate(16, (_) => rnd.nextInt(256));
    bytes[6] = (bytes[6] & 0x0f) | 0x40;
    bytes[8] = (bytes[8] & 0x3f) | 0x80;
    final String hex =
        bytes.map((int b) => b.toRadixString(16).padLeft(2, '0')).join();
    return '${hex.substring(0, 8)}-${hex.substring(8, 12)}-'
        '${hex.substring(12, 16)}-${hex.substring(16, 20)}-${hex.substring(20)}';
  }

  /// Production entry point: open the app-private file DB, apply the
  /// launch seed (C1), and hand back a ready store. The folder comes
  /// from path_provider, never a hard-coded or world-readable path.
  static Future<M1LibraryStore> openDefault() async {
    final Directory dir = await getApplicationSupportDirectory();
    final File file = File('${dir.path}/$kDbFileName');
    final M1Database db = M1Database(NativeDatabase.createInBackground(file));
    final M1LibraryStore store = M1LibraryStore(db);
    await store.seedIfEmpty();
    return store;
  }
